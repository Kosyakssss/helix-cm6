#!/usr/bin/env python3
"""Count lines, words and characters, like wc."""

import sys
from dataclasses import dataclass
from pathlib import Path


@dataclass
class Counts:
    lines: int = 0
    words: int = 0
    chars: int = 0

    def __add__(self, other: "Counts") -> "Counts":
        return Counts(
            self.lines + other.lines,
            self.words + other.words,
            self.chars + other.chars,
        )


def count(text: str) -> Counts:
    return Counts(text.count("\n"), len(text.split()), len(text))


def main(paths: list[str]) -> int:
    total = Counts()
    for name in paths:
        path = Path(name)
        if not path.exists():
            print(f"{name}: no such file", file=sys.stderr)
            continue
        counts = count(path.read_text(encoding="utf-8"))
        total += counts
        print(f"{counts.lines:>8}{counts.words:>8}{counts.chars:>8} {name}")
    if len(paths) > 1:
        print(f"{total.lines:>8}{total.words:>8}{total.chars:>8} total")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
